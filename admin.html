// فرض کنید متغیر disabledSlots شامل آرایه‌ای از کلیدها مثل ["1405-06-13_12:00", ...] است // و selectedDate تاریخ انتخابی ادمین است (مثلاً "1405-06-13") function renderAdminSlots(workingHours, disabledSlots, selectedDate) { const container = document.getElementById('slots-container'); // کانتینر دکمه‌های سانس در پنل container.innerHTML = ''; // ترکیب شیفت صبح و عصر به صورت پله‌های ۳۰ دقیقه‌ای استاندارد // یا می‌توانید مستقیماً از آرایه workingHours که شامل "12:00", "12:30" است استفاده کنید let allSlots = [...(workingHours.morning || []), ...(workingHours.evening || [])]; allSlots.forEach(time => { let slotKey = `${selectedDate}_${time}`; let isDisabled = disabledSlots.includes(slotKey); // محاسبه ساعت پایان برای نمایش زیبا (مثلا 12:00 تا 12:30) let [h, m] = time.split(':').map(Number); let endTotal = h * 60 + m + 30; let endH = Math.floor(endTotal / 60).toString().padStart(2, '0'); let endM = (endTotal % 60).toString().padStart(2, '0'); let timeRangeText = `${time} - ${endH}:${endM}`; let btn = document.createElement('button'); btn.className = `slot-btn ${isDisabled ? 'disabled' : 'active'}`; btn.innerHTML = `${timeRangeText} ${isDisabled ? '❌' : '✔️'}`; // قابلیت کلیک برای بستن یا باز کردن دستی سانس توسط ادمین btn.onclick = () => { toggleSlotStatus(slotKey); }; container.appendChild(btn); }); }
